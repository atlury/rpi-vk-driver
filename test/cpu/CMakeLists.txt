add_executable(cpu-regressions regressions.c
    ../../driver/map.c ../../driver/PoolAllocator.c ../../driver/ConsecutivePoolAllocator.c)
target_include_directories(cpu-regressions PRIVATE ${CMAKE_SOURCE_DIR}/driver)
add_dependencies(cpu-regressions vulkan-headers)
target_compile_options(cpu-regressions PRIVATE -Wall -Wextra -Werror)
option(CPU_TEST_SANITIZERS "Enable address and undefined behavior checks on CPU regressions" OFF)
if(CPU_TEST_SANITIZERS)
    target_compile_options(cpu-regressions PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cpu-regressions PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME cpu-regressions COMMAND cpu-regressions)

    add_test(NAME no-vc4 COMMAND no-vc4 $<TARGET_FILE:rpi-vk-driver>)
